# TinyTensor API contracts (C++17)

These declarations are the locked public core for the first implementation. They intentionally model straight-line SSA, rather than LLVM's full CFG/function hierarchy. Values and instructions are owned by `Module`; operands/usages are non-owning links. `Function` is deliberately absent until control flow is a real requirement.

## Required header: `include/tinytensor/ir.hpp`

```cpp
#pragma once
#include <cstddef>
#include <cstdint>
#include <memory>
#include <string>
#include <unordered_set>
#include <utility>
#include <vector>

namespace tinytensor {

enum class DType : std::uint8_t { F32 };

struct TensorType final {
  DType dtype{DType::F32};
  std::vector<std::int64_t> shape;  // scalar: {}; tensor rank is 1 or 2.
  bool isScalar() const noexcept { return shape.empty(); }
  std::size_t rank() const noexcept { return shape.size(); }
  bool operator==(const TensorType& other) const noexcept;
  bool operator!=(const TensorType& other) const noexcept { return !(*this == other); }
};

enum class Op : std::uint8_t {
  Input, Constant,
  Add, Sub, Mul, Div, Max,
  Relu, Exp, Neg,
  MatMul,
  FusedElementwise, FusedMatMulEpilogue
};

class Instruction;
class Module;

class Value final {
 public:
  using Id = std::uint32_t;
  Value(Id id, TensorType type, std::string name, Instruction* defining_inst) noexcept;
  Id id() const noexcept;
  const TensorType& type() const noexcept;
  const std::string& name() const noexcept;
  Instruction* definingInst() const noexcept; // null only for Module input/constant values.
  const std::vector<Instruction*>& users() const noexcept;
  std::size_t useCount() const noexcept;
  bool isOutput() const noexcept;

 private:
  friend class Module;
  friend class Instruction;
  Id id_; TensorType type_; std::string name_; Instruction* defining_inst_;
  std::vector<Instruction*> users_; bool is_output_{false};
  void addUser(Instruction* user);
  void removeUser(Instruction* user);
};

class Instruction final {
 public:
  Instruction(Op op, std::vector<Value*> operands, Value* result) noexcept;
  Op op() const noexcept;
  const std::vector<Value*>& operands() const noexcept;
  Value& result() const noexcept;
  void setOperand(std::size_t index, Value* replacement);

 private:
  friend class Module;
  Op op_; std::vector<Value*> operands_; Value* result_;
};

class Module final {
 public:
  Module() = default;
  Value& addInput(std::string name, TensorType type);
  Value& addConstant(float value); // scalar F32; canonical name generated by Module.
  Instruction& addInstruction(Op op, std::vector<Value*> operands,
                              TensorType result_type, std::string result_name = {});
  void addOutput(Value& value);
  bool eraseInstruction(Instruction& instruction); // false if its result still has users/is output.
  void replaceAllUsesWith(Value& from, Value& to);
  const std::vector<std::unique_ptr<Value>>& values() const noexcept;
  const std::vector<std::unique_ptr<Instruction>>& instructions() const noexcept;
  const std::vector<Value*>& inputs() const noexcept;
  const std::vector<Value*>& outputs() const noexcept;
  std::string print() const;

 private:
  Value::Id next_value_id_{0};
  std::vector<std::unique_ptr<Value>> values_;
  std::vector<std::unique_ptr<Instruction>> instructions_;
  std::vector<Value*> inputs_, outputs_;
};

class IRBuilder final {
 public:
  explicit IRBuilder(Module& module) noexcept;
  Value& input(std::string name, TensorType type);
  Value& constant(float value);
  Value& createUnary(Op op, Value& operand, std::string name = {});
  Value& createBinary(Op op, Value& lhs, Value& rhs, std::string name = {});
  Value& createMatMul(Value& lhs, Value& rhs, std::string name = {});
  void output(Value& value);
 private: Module& module_;
};

} // namespace tinytensor
```

`addInstruction` is the only normal constructor path for non-leaf values; it creates the result and immediately records reciprocal use-def links. `eraseInstruction` must remove its operand-use links and owns deletion. Do not store raw pointers outside the module lifetime.

## Required header: `include/tinytensor/pass.hpp`

```cpp
#pragma once
#include <memory>
#include <string>
#include <vector>
namespace tinytensor {
class Module;
class Pass {
 public:
  virtual ~Pass() = default;
  virtual std::string name() const = 0;
  virtual bool runOnModule(Module& module) = 0; // true iff module changed
};
class PassManager final {
 public:
  explicit PassManager(bool verify_after_each_pass = true) noexcept;
  void addPass(std::unique_ptr<Pass> pass);
  bool run(Module& module); // throws std::runtime_error if enabled verification fails
 private:
  bool verify_after_each_pass_; std::vector<std::unique_ptr<Pass>> passes_;
};
} // namespace tinytensor
```

## Required header: `include/tinytensor/verifier.hpp`

```cpp
#pragma once
#include <string>
namespace tinytensor { class Module;
struct VerificationResult final { bool ok; std::string message; };
VerificationResult verifyModule(const Module& module);
} // namespace tinytensor
```

Pass interfaces belong in separate headers, such as `passes/dce.hpp`: `class DCEPass final : public Pass { public: std::string name() const override; bool runOnModule(Module&) override; };` Fusion must additionally expose `FusionPass` and no mutable global state.

## Invariants enforced by the verifier

- IDs are unique and assigned in construction order; every operand/result pointer belongs to this module.
- Inputs/constants have null definitions; all other values have exactly one defining instruction.
- Every instruction result points back to that instruction; operand user lists and operand arrays agree exactly (including multiplicity).
- `Input`/`Constant` are not emitted as ordinary instructions; operations have the arity shown by `Op`.
- Shapes follow the scalar and rank-1 bias broadcast rules; `MatMul [M,K] × [K,N] -> [M,N]`.
- Output values belong to the module. Straight-line construction order supplies dominance: an operand definition must precede its user.

No pass may mutate an instruction's `Op` or `result` pointer directly; rebuild/replace through `Module` when necessary.
